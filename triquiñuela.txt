-- VISTA SIMPLE

CREATE OR REPLACE VIEW V_MES_PRIMER_SEMESTRE AS
    SELECT id, descripcion FROM MES WHERE id < 7;
    
    
-- VER CAMPOS DE LA VISTA    
SELECT descripcion  FROM v_mes_primer_semestre;

-- VER LA DESCRIPCION DE LA VISTA
DESC v_mes_primer_semestre;


-- VISTA COMPLEJA
--------------------------
CREATE OR REPLACE VIEW V_REVISTA_MES_ENERO AS
    SELECT d.TITULO AS "TITULO REVISTA"  FROM DOCUMENTO d INNER JOIN revista r ON d.codigo = r.codigo INNER JOIN mes m ON r.mes_id = m.id WHERE m.id = 1;
    
    
    
SELECT * FROM v_revista_mes_enero;


-- AÑADIR RESTRICCIONES AL USO DE LAS VISTAS
-- WITH CHECK OPTION : SE DEBE CUMPLIR LA CONDICIÓN DE LA VISTA (SOLO SE PUEDE MANIPULAR DATOS QUE CUMPLAN)
-- WITH READ ONLY : SOLO LECTURA

-- EJEMPLO WITH CHECK OPTION 
-- FORCE ES OPCIONAL POR DEFECTO ES FORCE
CREATE OR REPLACE FORCE VIEW V_MES_SEGUNDO_SEMESTRE AS
    SELECT id, descripcion FROM MES WHERE id > 6
    WITH CHECK OPTION CONSTRAINT CONTROL_SEMESTRE2; -- NOMBRE DE LA RESTRICCIÓN
    

-- EJEMPLO READ ONLY
-- NO DEJA ALTERAR DATOS
CREATE OR REPLACE VIEW V_DOCUMENTOS AS
    SELECT CODIGO, TITULO, AUTOR, EDITORIAL, ANIO FROM DOCUMENTO
    WITH READ ONLY CONSTRAINT CONTROL_DOCS;
